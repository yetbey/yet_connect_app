create table public.notifications (
  id bigint generated by default as identity not null,
  type text not null,
  message text not null,
  is_read boolean null default false,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  receiver_id uuid not null,
  sender_id uuid null,
  post_id bigint null,
  constraint notifications_pkey primary key (id),
  constraint notifications_post_id_fkey foreign KEY (post_id) references posts (id) on delete CASCADE,
  constraint notifications_receiver_id_fkey foreign KEY (receiver_id) references profiles (id) on delete CASCADE,
  constraint notifications_sender_id_fkey foreign KEY (sender_id) references profiles (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_notifications_receiver on public.notifications using btree (receiver_id, created_at desc) TABLESPACE pg_default;
