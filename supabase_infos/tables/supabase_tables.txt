// App Verions Table
create table public.app_versions (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  version text not null,
  download_url text not null,
  is_mandatory boolean null default false,
  constraint app_versions_pkey primary key (id)
) TABLESPACE pg_default;

// Chat Participants Table
create table public.chat_participants (
  id bigint generated by default as identity not null,
  chat_id bigint not null,
  user_id uuid not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  deleted_at timestamp with time zone null,
  constraint chat_participants_pkey primary key (id),
  constraint chat_participants_chat_id_user_id_key unique (chat_id, user_id),
  constraint chat_participants_chat_id_fkey foreign KEY (chat_id) references chats (id) on delete CASCADE,
  constraint chat_participants_user_id_fkey foreign KEY (user_id) references profiles (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_chat_participants_deleted_at on public.chat_participants using btree (deleted_at) TABLESPACE pg_default;

// Chat Table
create table public.chats (
  id bigint generated by default as identity not null,
  last_message text null,
  last_message_at timestamp with time zone not null default timezone ('utc'::text, now()),
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  constraint chats_pkey primary key (id)
) TABLESPACE pg_default;

// Comments Table
create table public.comments (
  id bigint generated by default as identity not null,
  content text not null,
  user_id uuid not null,
  post_id bigint not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  constraint comments_pkey primary key (id),
  constraint comments_post_id_fkey foreign KEY (post_id) references posts (id) on delete CASCADE,
  constraint comments_user_id_fkey foreign KEY (user_id) references profiles (id)
) TABLESPACE pg_default;

create index IF not exists idx_comments_post_id on public.comments using btree (post_id) TABLESPACE pg_default;

create trigger on_new_comment
after INSERT on comments for EACH row
execute FUNCTION handle_new_comment ();

// Follows Table
create table public.follows (
  follower_id uuid not null,
  following_id uuid not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  constraint follows_pkey primary key (follower_id, following_id),
  constraint follows_follower_id_fkey foreign KEY (follower_id) references profiles (id),
  constraint follows_following_id_fkey foreign KEY (following_id) references profiles (id)
) TABLESPACE pg_default;

create index IF not exists idx_follows_follower_following on public.follows using btree (follower_id, following_id) TABLESPACE pg_default;

create trigger on_new_follow
after INSERT on follows for EACH row
execute FUNCTION handle_new_follow ();

// Login Sessions Table

  session_token text not null,
  user_id uuid null,
  is_authenticated boolean null default false,
  device_info text null,
  created_at timestamp with time zone null default timezone ('utc'::text, now()),

  constraint login_sessions_pkey primary key (id),
  constraint login_sessions_session_token_key unique (session_token),
  constraint login_sessions_user_id_fkey foreign KEY (user_id) references profiles (id)

create index IF not exists idx_login_sessions_token on public.login_sessions using btree (session_token) TABLESPACE pg_default;

create index IF not exists idx_login_sessions_expires on public.login_sessions using btree (expires_at) TABLESPACE pg_default;

// Messages Table
create table public.messages (
  id bigint generated by default as identity not null,
  content text null,
  image_url text null,
  sender_id uuid not null,
  receiver_id uuid not null,
  is_read boolean null default false,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  chat_id bigint null,
  reply_to_id bigint null,
  reply_to_message_id bigint null,
  reply_to_content text null,
  reply_to_image_url text null,
  reply_to_sender_name text null,
  constraint messages_pkey primary key (id),
  constraint messages_chat_id_fkey foreign KEY (chat_id) references chats (id) on delete CASCADE,
  constraint messages_receiver_id_fkey foreign KEY (receiver_id) references profiles (id),
  constraint messages_reply_to_id_fkey foreign KEY (reply_to_id) references messages (id) on delete set null,
  constraint messages_reply_to_message_id_fkey foreign KEY (reply_to_message_id) references messages (id) on delete set null,
  constraint messages_sender_id_fkey foreign KEY (sender_id) references profiles (id)
) TABLESPACE pg_default;

create index IF not exists idx_messages_reply on public.messages using btree (reply_to_message_id) TABLESPACE pg_default;

// Notifications Table
create table public.notifications (
  id bigint generated by default as identity not null,
  type text not null,
  message text not null,
  is_read boolean null default false,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  receiver_id uuid not null,
  sender_id uuid null,
  post_id bigint null,
  constraint notifications_pkey primary key (id),
  constraint notifications_post_id_fkey foreign KEY (post_id) references posts (id) on delete CASCADE,
  constraint notifications_receiver_id_fkey foreign KEY (receiver_id) references profiles (id) on delete CASCADE,
  constraint notifications_sender_id_fkey foreign KEY (sender_id) references profiles (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_notifications_receiver on public.notifications using btree (receiver_id, created_at desc) TABLESPACE pg_default;

// Posts Table
create table public.posts (
  id bigint generated by default as identity not null,
  caption text null,
  image_url text null,
  user_id uuid not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  video_url text null,
  alignment_y double precision null default '0'::double precision,
  alignment_x double precision null default '0'::double precision,
  tags text[] null default '{}'::text[],
  constraint posts_pkey primary key (id),
  constraint posts_user_id_fkey foreign KEY (user_id) references profiles (id)
) TABLESPACE pg_default;

create index IF not exists idx_posts_tags on public.posts using gin (tags) TABLESPACE pg_default;

create index IF not exists idx_posts_user_created on public.posts using btree (user_id, created_at desc) TABLESPACE pg_default;

create index IF not exists idx_posts_video_null_created on public.posts using btree (created_at desc) TABLESPACE pg_default
where
  (video_url is null);

create trigger trigger_update_tag_counts
after INSERT
or DELETE
or
update on posts for EACH row
execute FUNCTION update_tag_counts ();

// Profiles Table
create table public.profiles (
  id uuid not null,
  username text null,
  full_name text null,
  bio text null,
  profile_image_url text null,
  phone_number text null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  fcm_token text null,
  constraint profiles_pkey primary key (id),
  constraint profiles_username_key unique (username),
  constraint profiles_id_fkey foreign KEY (id) references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

// Tags Table
create table public.tags (
  id bigint generated by default as identity not null,
  name text not null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  post_count integer null default 0,
  updated_at timestamp with time zone null default now(),
  constraint tags_pkey primary key (id),
  constraint tags_name_key unique (name)
