
BEGIN
  RETURN QUERY
  SELECT
    p.id,
    p.caption,
    p.image_url,
    p.video_url,
    p.user_id,
    p.created_at,
    p.alignment_x,
    p.alignment_y,
    p.tags,
    prof.username,
    prof.full_name,
    prof.profile_image_url,
    COALESCE(COUNT(DISTINCT pl.user_id), 0)::bigint as like_count,
    COALESCE(COUNT(DISTINCT c.id), 0)::bigint as comment_count,
    BOOL_OR(pl.user_id = p_user_id) as is_liked_by_user
  FROM posts p
  INNER JOIN profiles prof ON prof.id = p.user_id
  LEFT JOIN post_likes pl ON pl.post_id = p.id
  LEFT JOIN comments c ON c.post_id = p.id
  WHERE
    p.video_url IS NULL
    AND (p_last_post_timestamp IS NULL OR p.created_at < p_last_post_timestamp)
    AND (
      NOT p_only_following
      OR p.user_id IN (
        SELECT following_id FROM follows WHERE follower_id = p_user_id
        UNION ALL
        SELECT p_user_id
      )
    )
  GROUP BY p.id, prof.username, prof.full_name, prof.profile_image_url
  ORDER BY p.created_at DESC
  LIMIT p_limit;
END;
