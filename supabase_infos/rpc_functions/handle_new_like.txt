
begin
  -- Kendi gönderisini beğendiyse bildirim atma
  if new.user_id = (select user_id from posts where id = new.post_id) then
    return new;
  end if;

  insert into public.notifications (type, sender_id, receiver_id, post_id, message)
  values (
    'like',
    new.user_id,
    (select user_id from posts where id = new.post_id),
    new.post_id,
    'gönderini beğendi.'
  );
  return new;
end;
