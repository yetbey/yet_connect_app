
declare
  current_user_id uuid;
  existing_chat_id bigint; -- Değişken tipi bigint
  new_chat_id bigint; -- Değişken tipi bigint
begin
  current_user_id := auth.uid();

  -- İki kullanıcının ortak olduğu bir chat_id var mı?
  select cp1.chat_id into existing_chat_id
  from chat_participants cp1
  join chat_participants cp2 on cp1.chat_id = cp2.chat_id
  where cp1.user_id = current_user_id
  and cp2.user_id = target_user_id
  limit 1;

  -- Varsa döndür
  if existing_chat_id is not null then
    return existing_chat_id;
  end if;

  -- Yoksa yeni oluştur (chats tablosu bigint id üretiyor)
  insert into chats (created_at)
  values (now())
  returning id into new_chat_id;

  -- Katılımcıları ekle
  insert into chat_participants (chat_id, user_id)
  values 
    (new_chat_id, current_user_id),
    (new_chat_id, target_user_id);

  return new_chat_id;
end;
